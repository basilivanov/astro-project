                                             REPO_ROOT --worktree-root
                                             WORKTREE_ROOT --worktree-path
                                             WORKTREE_PATH --project-key
                                             PROJECT_KEY --packet-id PACKET_ID

                                             BASE_REF [--remote REMOTE]

options:
  -h, --help            show this help message and exit
  --packet PACKET       Path to EXECUTION_PACKET.md
  --repo-root REPO_ROOT
                        Main repository root
  --worktree-root WORKTREE_ROOT
                        Allowed worktree root
  --worktree-path WORKTREE_PATH
                        Packet worktree path
  --project-key PROJECT_KEY
                        Project key
  --packet-id PACKET_ID
                        Packet ID
  --attempt ATTEMPT     Attempt number
  --base-ref BASE_REF   Base ref used for the packet branch
  --remote REMOTE       Remote name for packet branch push
  --dry-run             Plan only; default when --apply is omitted
  --apply               Allow requested Git commit/push mutations
  --commit              Request guarded packet worktree commit
  --push                Request guarded packet branch push
  --allow-git-commit    Approve commit application
  --allow-git-push      Approve push application
  --json                JSON output
